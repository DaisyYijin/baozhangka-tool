## 🚀 使用示例

### 1. 基础使用
```html
<script type="module">
    import CardRegistry from './scripts/发卡收卡登记/index.js';
    
    // 初始化
    CardRegistry.init();
    
    // 添加发卡记录
    const issueRecord = CardRegistry.createIssueRecord({
        name: '张三',
        idNumber: '110101199001011234',
        department: '第一连',
        cardType: 'Ⅰ类卡',
        cardNumber: 'BZK20240001',
        issueDate: '2024-01-01'
    });
    CardRegistry.addRecord(issueRecord);
    
    // 渲染表格
    CardRegistry.renderTable();
</script>
```

### 2. 搜索和过滤
```javascript
// 搜索姓名
CardRegistry.searchRecords('张三');

// 按卡类型过滤
CardRegistry.filterByCardType('Ⅰ类卡');

// 按年份过滤
CardRegistry.filterByYear('2024');

// 获取过滤结果
const results = CardRegistry.getFilteredRecords();
console.log('找到', results.length, '条记录');

// 渲染结果
CardRegistry.renderTable();
```

### 3. 导入导出
```javascript
// 导出到Excel
CardRegistry.exportToExcel();

// 下载模板
CardRegistry.downloadTemplate();

// 导入（配合文件输入）
const fileInput = document.getElementById('importFile');
fileInput.addEventListener('change', async (e) => {
    const file = e.target.files[0];
    try {
        const count = await CardRegistry.importFromExcel(file);
        alert('成功导入 ' + count + ' 条记录');
        CardRegistry.renderTable();
    } catch (err) {
        alert('导入失败: ' + err.message);
    }
});

// 备份数据
CardRegistry.backupCompleteData();  // 全部数据
CardRegistry.backupAllData();        // 当前类型
```

### 4. 事件处理
```javascript
// 初始化所有事件监听器
CardRegistry.initEventListeners();

// 手动绑定特定事件
document.getElementById('searchInput').addEventListener('input', 
    CardRegistry.handleSearchInput
);

// 切换标签
CardRegistry.switchSubTab('issue');    // 发卡标签
CardRegistry.switchSubTab('recycle');  // 收卡标签
```
